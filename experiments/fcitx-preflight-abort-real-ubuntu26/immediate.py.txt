import asyncio, json, time
from dbus_next import Message
from inkecho.platform.fcitx5 import Fcitx5Bridge, FcitxOutputError

async def main():
    outcomes=[]
    for index in range(30):
        first=Fcitx5Bridge()
        second=Fcitx5Bridge()
        start=time.monotonic()
        owner_present=None
        cap_closed=None
        second_elapsed=None
        try:
            await first.connect()
            old_owner=first._bus.unique_name
            cap=first._capability
            await first.abort_preflight()
            cap_closed=(cap.token=='' and cap.fd is None and cap.lock_fd is None)
            await second.connect()
            second_elapsed=round(time.monotonic()-start,6)
            answer=await second._bus.call(Message(destination='org.freedesktop.DBus',path='/org/freedesktop/DBus',interface='org.freedesktop.DBus',member='NameHasOwner',signature='s',body=[old_owner]))
            owner_present=answer.body[0]
            result='success'
        except FcitxOutputError as exc:
            result=exc.code
        except Exception as exc:
            result=type(exc).__name__
        finally:
            await first.aclose()
            await second.aclose()
        outcomes.append({'index':index,'result':result,'second_ready_s':second_elapsed,'old_owner_present':owner_present,'old_capability_closed':cap_closed})
    print(json.dumps({'outcomes':outcomes,'successes':sum(x['result']=='success' for x in outcomes),'attempts':len(outcomes),'commits_requested':0}),flush=True)
asyncio.run(main())
