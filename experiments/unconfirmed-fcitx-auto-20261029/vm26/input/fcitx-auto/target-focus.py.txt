    events_out=result.get('events',[])
    if os.environ.get('INK_TARGET_KIND')=='owner':
        route=(['backend','fcitx'] in events_out and any(e[0]=='owner_replaced' for e in events_out) and any(e[:2]==['fcitx_error','owner_changed'] for e in events_out) and ['keyboard_insert_started'] not in events_out and sum(e==['commit_status',1] for e in events_out)==1)
    elif os.environ.get('INK_TARGET_KIND')=='normal':
        route=(['backend','fcitx'] in events_out and ['commit_status',1] in events_out and ['keyboard_insert_started'] not in events_out and not any(e[0]=='fcitx_error' for e in events_out))
    else:
        route=(['backend','fcitx'] in events_out and ['fcitx_error','no_target',0,0,False] in events_out and ['keyboard_insert_started'] in events_out and events_out.index(['fcitx_error','no_target',0,0,False]) < events_out.index(['keyboard_insert_started']))
