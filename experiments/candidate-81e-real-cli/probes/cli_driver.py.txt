import json, os, signal, subprocess, tempfile, time
from pathlib import Path
root=Path('/tmp/inkecho-r4-test'); expected='风云日月山川河海'*25
with tempfile.TemporaryDirectory(prefix='inkecho-cli-') as d:
 out=Path(d)/'cat-output'; pidfile=Path(d)/'cat-pid'
 term=subprocess.Popen(['/usr/bin/gnome-terminal','--wait','--title','InkEcho isolated cat','--','/usr/bin/bash','-c','printf "%s" "$$" > "$2"; stty raw -echo; exec /usr/bin/cat > "$1"','_',str(out),str(pidfile)],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
 for _ in range(200):
  if out.exists() and pidfile.exists(): break
  if term.poll() is not None: break
  time.sleep(.05)
 if not out.exists(): print(json.dumps({'error':'cat_not_ready'})); raise SystemExit(1)
 time.sleep(2)
 app=root/'squashfs-root'; env=os.environ.copy();env.update(PYTHONHOME=str(app/'usr'),PYTHONPATH=str(app/'packages'),LD_LIBRARY_PATH=str(app/'usr/lib/inkecho'),QT_QPA_PLATFORM='offscreen',INK_TARGET_KIND='normal',INK_PROBE_MODE=os.environ.get('INK_CLI_MODE','auto'))
 sender=subprocess.run([str(app/'usr/bin/python3'),str(root/'probes/sender.py')],env=env,capture_output=True,text=True,timeout=45)
 time.sleep(1); raw=out.read_bytes();text=raw.decode('utf8',errors='replace')
 print(json.dumps({'mode':env['INK_PROBE_MODE'],'exit':sender.returncode,'received':text,'received_bytes_hex':raw.hex(),'length':len(text),'exact':text==expected,'stdout':sender.stdout,'stderr':sender.stderr[-3000:]},ensure_ascii=False),flush=True)
 try: os.kill(int(pidfile.read_text()),signal.SIGTERM)
 except (OSError,ValueError): pass
 term.terminate()
 try: term.wait(timeout=3)
 except subprocess.TimeoutExpired: term.kill();term.wait()
