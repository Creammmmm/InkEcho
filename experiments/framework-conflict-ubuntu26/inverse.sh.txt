set -eu
root=/tmp/inkecho-r4-test; APPDIR=$root/squashfs-root
export XDG_RUNTIME_DIR=/run/user/1000 DBUS_SESSION_BUS_ADDRESS=unix:path=/run/user/1000/bus DISPLAY=:0 XDG_SESSION_TYPE=wayland XDG_CURRENT_DESKTOP=ubuntu:GNOME
old=$(mktemp); systemctl --user show-environment | sed -n '/^XMODIFIERS=/p; /^GTK_IM_MODULE=/p; /^QT_IM_MODULE=/p' > "$old"
cleanup(){ systemctl --user unset-environment XMODIFIERS GTK_IM_MODULE QT_IM_MODULE || true; while IFS= read -r item; do systemctl --user set-environment "$item" || true; done < "$old"; rm -f "$old"; }
trap cleanup EXIT
probe(){ env LABEL="$1" PYTHONHOME="$APPDIR/usr" PYTHONPATH="$APPDIR/packages" LD_LIBRARY_PATH="$APPDIR/usr/lib/inkecho" "$APPDIR/usr/bin/python3" "$root/probes/probe.py"; }
systemctl --user set-environment XMODIFIERS=@im=ibus GTK_IM_MODULE=ibus QT_IM_MODULE=ibus
probe coherent_ibus
systemctl --user set-environment XMODIFIERS=@im=fcitx GTK_IM_MODULE=fcitx QT_IM_MODULE=fcitx
probe stale_fcitx_hints
systemctl --user set-environment XMODIFIERS=@im=ibus GTK_IM_MODULE=fcitx QT_IM_MODULE=ibus
probe mixed_hints_ibus
