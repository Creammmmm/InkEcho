import json,os,signal,subprocess,sys,tempfile,time
from pathlib import Path
mode=sys.argv[1]; root=Path('/tmp/inkecho-r4-test'); expected='风云日月山川河海'*25
with tempfile.TemporaryDirectory() as d:
 p=Path(d); out=p/'a'; second=p/'b'; terminals=[]
 def launch(out,title):
  proc=subprocess.Popen(['/usr/bin/gnome-terminal','--wait','--title',title,'--','/usr/bin/python3','-u',str(root/'probes/reader.py'),str(out)],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
  terminals.append(proc)
  for _ in range(200):
   if out.exists():return True
   if proc.poll() is not None:return False
   time.sleep(.05)
  return False
 def read(path):
  try:return json.loads(path.read_text())
  except (OSError,ValueError):return ''
 if not launch(out,'InkEcho isolated terminal A'):
  print(json.dumps({'mode':mode,'error':'terminal_a_not_ready'}),flush=True);sys.exit(1)
 time.sleep(2)
 app=root/'squashfs-root'; env=os.environ.copy();env.update(PYTHONHOME=str(app/'usr'),PYTHONPATH=str(app/'packages'),LD_LIBRARY_PATH=str(app/'usr/lib/inkecho'),QT_QPA_PLATFORM='offscreen',INK_TARGET_KIND='normal',INK_PROBE_MODE='ibus')
 with tempfile.TemporaryFile(mode='w+') as stdout,tempfile.TemporaryFile(mode='w+') as stderr:
  sender=subprocess.Popen([str(app/'usr/bin/python3'),str(root/'probes/sender.py')],env=env,stdout=stdout,stderr=stderr)
  cancelled=switched=resumed=False; at_cancel=None; end=time.monotonic()+40
  while sender.poll() is None and time.monotonic()<end:
   a=read(out)
   if a and mode=='cancel' and not cancelled:cancelled=True;at_cancel=a;os.kill(sender.pid,signal.SIGUSR1)
   if a and mode=='switch' and not switched:
    switched=True;os.kill(sender.pid,signal.SIGSTOP)
    if launch(second,'InkEcho isolated terminal B'):
     time.sleep(2);resumed=True;os.kill(sender.pid,signal.SIGCONT)
   time.sleep(.01)
  if sender.poll() is None:
   if switched and not resumed:os.kill(sender.pid,signal.SIGCONT)
   sender.terminate();sender.wait(timeout=3)
  time.sleep(1);a=read(out);b=read(second);stdout.seek(0);stderr.seek(0)
  print(json.dumps({'mode':mode,'exit':sender.returncode,'a':a,'b':b,'lengths':[len(a),len(b)],'exact':a+b==expected,'prefix':expected.startswith(a+b),'cancel_sent':cancelled,'text_at_cancel':at_cancel,'switched':switched,'resumed':resumed,'stdout':stdout.read(),'stderr':stderr.read()[-3000:]},ensure_ascii=False),flush=True)
 for file in (out,second):
  try:os.kill(int(Path(str(file)+'.pid').read_text()),signal.SIGTERM)
  except (OSError,ValueError):pass
 for proc in terminals:
  proc.terminate()
  try:proc.wait(timeout=3)
  except subprocess.TimeoutExpired:proc.kill();proc.wait()
