#!/bin/bash
set -eu
if [ "${ISOLATED_BUS:-}" != 1 ]; then exec dbus-run-session -- env ISOLATED_BUS=1 bash "$0"; fi
root=$(mktemp -d /tmp/inkecho-presend.XXXXXX)
chmod 700 "$root"
export XDG_RUNTIME_DIR="$root" GIO_USE_VFS=local
trap 'rm -rf "$root"' EXIT
for mode in none partial lost; do
 echo "MODE=$mode"
 TEST_MODE="$mode" gjs -m /tmp/ibus-presend-real/private-service.js > "$root/service-$mode.log" 2>&1 & pid=$!
 for i in $(seq 1 100); do grep -q ready "$root/service-$mode.log" && break; sleep .05; done
 TEST_MODE="$mode" PYTHONPATH=/home/lth/github/InkEcho/src /home/lth/github/InkEcho/.venv/bin/python /tmp/ibus-presend-real/private-client.py
 kill "$pid";wait "$pid" || true
 cat "$root/service-$mode.log"
done
