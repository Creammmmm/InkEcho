set -eu
umask 077
root=/tmp/inkecho-r4-test
cd "$root"
printf '%s  %s\n' "$1" "$root/InkEcho-0.10.2-x86_64.AppImage" | sha256sum -c -
export XDG_RUNTIME_DIR=/run/user/1000 DBUS_SESSION_BUS_ADDRESS=unix:path=/run/user/1000/bus DISPLAY=:0 XMODIFIERS=@im=fcitx QT_IM_MODULE=fcitx
export XDG_SESSION_TYPE="$2" XDG_CURRENT_DESKTOP=ubuntu:GNOME INK_PROBE_MODE=auto
if [ "$2" = wayland ]; then export GDK_BACKEND=wayland WAYLAND_DISPLAY=wayland-0; else export GDK_BACKEND=x11; fi
export XAUTHORITY="$(systemctl --user show-environment | sed -n 's/^XAUTHORITY=//p')"
test -n "$XAUTHORITY" && test -r "$XAUTHORITY"
! pgrep -x fcitx5 >/dev/null
base=$(mktemp -d /tmp/inkecho-r4-fcitx.XXXXXX)
mkdir -p "$base/share" "$base/config" "$base/cache"
export XDG_DATA_HOME="$base/share" XDG_CONFIG_HOME="$base/config" XDG_CACHE_HOME="$base/cache"
export APPDIR="$root/squashfs-root"
unset FCITX_ADDON_DIRS PYTHONHOME PYTHONPATH LD_LIBRARY_PATH QT_QPA_PLATFORM
systemctl --user show-environment | sed -n '/^XMODIFIERS=/p; /^GTK_IM_MODULE=/p; /^QT_IM_MODULE=/p' > "$base/oldenv"
cleanup() {
 systemctl --user unset-environment XMODIFIERS GTK_IM_MODULE QT_IM_MODULE || true
 while IFS= read -r entry; do systemctl --user set-environment "$entry" || true; done < "$base/oldenv"
 test -z "${pid:-}" || kill "$pid" 2>/dev/null || true
 test -z "${pid:-}" || wait "$pid" 2>/dev/null || true
 rm -rf "$base"
}
trap cleanup EXIT
chmod +x "$root/InkEcho-0.10.2-x86_64.AppImage"
test -d "$APPDIR" || "$root/InkEcho-0.10.2-x86_64.AppImage" --appimage-extract >/dev/null
sha256sum "$APPDIR/packages/inkecho/platform/"{fcitx5,framework_detection,ibus}.py
systemctl --user set-environment XMODIFIERS=@im=fcitx GTK_IM_MODULE=fcitx QT_IM_MODULE=fcitx
fcitx5 --disable xim -D > "$root/probes/fcitx-$2.log" 2>&1 & pid=$!
sleep 3
kill -0 "$pid"
py(){ env LABEL="$1" PYTHONHOME="$APPDIR/usr" PYTHONPATH="$APPDIR/packages" LD_LIBRARY_PATH="$APPDIR/usr/lib/inkecho" "$APPDIR/usr/bin/python3" "$root/probes/probe.py"; }
manage(){ env PYTHONHOME="$APPDIR/usr" PYTHONPATH="$APPDIR/packages" LD_LIBRARY_PATH="$APPDIR/usr/lib/inkecho" "$APPDIR/usr/bin/python3" -c "from inkecho.platform.fcitx5_install import FcitxInstaller; m=FcitxInstaller(); m.$1(); print(m.status())"; }
live(){ /usr/bin/gdbus call --session --dest org.fcitx.Fcitx5 --object-path /inkecho --method org.fcitx.Fcitx.InkEcho1.RuntimeVersion || true; }
manage install
manage enable
echo before_manual; py before_manual; live
addons(){ /usr/bin/gdbus call --session --dest org.fcitx.Fcitx5 --object-path /controller --method org.fcitx.Fcitx.Controller1.GetAddonsV2 | tr "(" "\n" | grep "inkecho" || true; }
state(){ /usr/bin/gdbus call --session --dest org.fcitx.Fcitx5 --object-path /controller --method org.fcitx.Fcitx.Controller1.SetAddonsState "$1"; }
echo addon_before; addons
state "[('inkecho', false)]"
echo addon_after_disabled; addons; py manual_disabled; live
state "[('inkecho', true)]"
echo addon_after_enabled; addons; py manual_enabled; live
manage disable
